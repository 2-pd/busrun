███ manifest.json ███

instance_name インスタンスの表示名

manual_url ユーザーマニュアルのURL


███ agencies.json ███

{
    "agencies" : {
        "事業者識別名" : {
            "agency_name" : 事業者表示名,
            "base_color" : 事業者のベースカラーの色コード,
            "color_stripes" : [ ベースカラー以外の表示色情報を重なり順の上側から列挙(ベースカラー以外の表示色がない場合は省略可能)
                {
                    "color" : 色コード,
                    "start" : 開始点の数値(0〜100),
                    "end" : 終了点の数値(0〜100)
                }...
            ]
        }...
    },
    "categories" : [ 事業者グループ
        {
            "category_name" : グループ名,
            "category_color" : グループのテーマカラー,
            "agencies" : グループに属する事業者の識別名を配列で
        }...
    ]
}

███ agency_info.json ███

{
    "agency_name" : 事業者表示名,
    "base_color" : 事業者のベースカラーの色コード,
    "color_stripes" : [ ベースカラー以外の表示色情報を重なり順の上側から列挙(ベースカラー以外の表示色がない場合は省略可能)
        {
            "color" : 色コード,
            "start" : 開始点の数値(0〜100),
            "end" : 終了点の数値(0〜100)
        }...
    ],
    "description" : 事業者データの説明文(URLは表示時に自動でリンクに変換される。省略可能),
    "license_text" : ライセンス文(URLは表示時に自動でリンクに変換される。省略可能),
    "license_introduction" : ライセンス概要(省略可能),
    "related_links" : [ 関連リンク(省略可能)
        {
            "link_url" : リンクURL,
            "link_text" : リンクテキスト,
            "link_description" : リンクの説明(省略可能)
        }...
    ]
}

███ route_info.json ███

※このファイルは変換スクリプトにより自動生成されるものである
{
    "stops" : { 停留所の一覧
        "停留所識別名" : {
            "stop_name" : 停留所表示名,
            "stop_lat" : 停留所中心点の緯度,
            "stop_lon" : 停留所中心点の経度,
            "platforms" : { 乗り場の一覧
                "乗り場識別名" : {
                    "platform_code" : 乗り場番号または乗り場名,
                    "alighting_only" : 降車専用の乗り場か否か(乗車可能な乗り場の場合は省略可能)
                }...
            }
        }...
    },
    "routes" : { 系統の一覧
        "経路識別名" : {
            "route_number" : 系統番号(アルファベットや漢字を含めてもよい。ない場合はnull),
            "route_name" : 経路名,
            "route_color": 系統の背景色(ない場合はnull),
            "route_text_color" : 系統の文字色(ない場合はnull),
            "route_desc" : 備考,
            "stops" : [ 停留所の一覧(往復区分が0の場合での発着順に記載)
                {
                    "stop_id" : 停留所識別名,
                    "platform_id" : 乗り場識別名
                }...
            ]
        }...
    },
    "major_stops" : 主要停留所の停留所識別名を配列で
}


███ diagram_info.json ███

※このファイルの大部分は変換スクリプトにより自動生成されるが、一部は手動での入力が必要となる
{
    "services" : { ダイヤの一覧
        "ダイヤ識別名" : {
            "service_name" : ダイヤ表示名,
            "main_color" : 時刻表の背景色
        }
    },
    "service_order" : ダイヤ識別名を表示させたい順に配列で,
    "calendar" : [ ダイヤ判定規則
        {
            "service_id" : ダイヤ識別名,
            "start_date" : 「YYYY-MM-DD」形式の期間開始日,
            "end_date" : 「YYYY-MM-DD」形式の期間終了日,
            "implementation_days" : [ どの曜日に有効化するか(0:日曜→6:土曜)
                日曜日と祝日に有効化するか否か,
                月曜日に有効化するか否か,
                火曜日に有効化するか否か,
                水曜日に有効化するか否か,
                木曜日に有効化するか否か,
                金曜日に有効化するか否か,
                土曜日に有効化するか否か
            ],
        }...
    ],
    "calendar_dates" : { 例外ダイヤでの運転日
        "「YYYY-MM-DD」形式の実施日" : ダイヤ識別名
    }
}


███ operation_table_****.json ███

※ファイル名の「****」部分はダイヤ識別名が入る
※このファイルは変換スクリプトにより自動生成されるものである
{
    "operations" : {
        "車両行路識別名" : { 各車両行路の情報
            "trips" : [ 各便の情報
                {
                    "trip_id" : 便識別名,
                    "first_departure_time" : 「HH:MM」形式の始発時刻,
                    "final_arrival_time" : 「HH:MM」形式の終着時刻,
                    "starting_stop" : 始発停留所の停留所識別名,
                    "terminal_stop" : 終着停留所の停留所識別名
                }...
            ],
            "starting_time" : 「HH:MM」形式の出庫時刻,
            "ending_time" : 「HH:MM」形式の入庫時刻
        }...
    },
    "operation_order" : 車両行路識別名を表示順に配列で
}


███ timetable_****.json ███

※ファイル名の「****」部分はダイヤ識別名が入る
※このファイルは変換スクリプトにより自動生成されるものである
{
    "便識別名" : {
        "route_id" : 経路識別名,
        "direction_id" : 往復区分(真偽値。往復区分がない場合はfalseとする),
        "stop_headsigns" : [ 行き先表示情報
            {
                "stop_index" : 何番目に発着する停留所から表示される行き先か(系統の始発停留所を0番目と数える),
                "stop_headsign" : 行き先表示
            }...
        ],
        "departure_times" : 経路内の各停留所発車時刻を配列で(時刻は「HH:MM」形式。走行区間に含まれていない停留所はnull),
        "pickup_types" : 降車専用の停留所か否かを配列で(走行区間に含まれていない停留所はnull)
    }...
}
